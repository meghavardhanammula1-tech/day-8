import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt

# Dataset
data = pd.DataFrame({
    "x": np.arange(1, 11),
    "y": [5, 7, 8, 6, 9, 11, 10, 12, 14, 13],
    "category": ["A","A","B","B","A","B","A","B","A","B"]
})

# Seaborn Style
sns.set_style("darkgrid")

# Line Plot
sns.lineplot(
    data=data,
    x="x",
    y="y",
    color="blue",
    linewidth=3,
    marker="o"
)

plt.title("Line Plot")
plt.xlabel("X Values")
plt.ylabel("Y Values")
plt.show()

# Scatter Plot
sns.scatterplot(
    data=data,
    x="x",
    y="y",
    hue="category"
)

plt.title("Scatter Plot")
plt.show()

# Box Plot
sns.boxplot(
    data=data,
    x="category",
    y="y"
)

plt.title("Box Plot")
plt.show()

# Violin Plot
sns.violinplot(
    data=data,
    x="category",
    y="y"
)

plt.title("Violin Plot")
plt.show()

# Swarm Plot
sns.swarmplot(
    data=data,
    x="category",
    y="y"
)

plt.title("Swarm Plot")
plt.show()

# Bar Plot
sns.barplot(
    data=data,
    x="category",
    y="y"
)

plt.title("Bar Plot")
plt.show()

# Point Plot
sns.pointplot(
    data=data,
    x="category",
    y="y"
)

plt.title("Point Plot")
plt.show()

# KDE Plot
sns.kdeplot(
    data=data["y"],
    fill=True
)

plt.title("KDE Plot")
plt.show()
